rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // === PUBLIC ACCESS — Login check ===
    match /credentials/{role} {
      allow read: if true;
      allow write: if false; // Only SP can change via Console
    }

    // === PRIVACY PIN — SP only ===
    match /settings/privacyPin {
      allow read: if true;
      allow write: if false; // SP sets via Console
    }

    // === STORES — All authenticated can read, Admin write ===
    match /stores/{storeId} {
      allow read: if true;
      allow create: if request.auth != null;
      allow update, delete: if false; // Admin manages via logic
    }

    // === CASHIERS — Read for login, Admin write ===
    match /cashiers/{cashierId} {
      allow read: if true;
      allow create: if false; // Admin only
    }

    // === PRODUCTS ===
    match /products/{productId} {
      allow read: if true;
      allow create: if false; // Admin only
    }

    // === TRANSACTIONS / SALES ===
    match /transactions/{txId} {
      allow read: if true;
      allow create: if request.auth != null;
      allow update: if false;
    }

    // === SHIFTS ===
    match /shifts/{shiftId} {
      allow read: if true;
      allow create: if request.auth != null;
      allow update: if false;
    }

    // === MESSAGES / CHAT ===
    match /messages/{msgId} {
      allow read: if true;
      allow create: if request.auth != null;
    }

    // === SUBSCRIPTION / SETTINGS ===
    match /subscription/{docId} {
      allow read: if true;
      allow write: if false; // SP only
    }
  }
}
